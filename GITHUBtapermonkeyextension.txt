// ==UserScript==
// @name         GitHub Repo Uploader — folders & files
// @namespace    https://github.com/
// @version      1.0.0
// @description  Add files or whole folders to any GitHub repo. Drag & drop, pick from disk, create new folders, and commit everything in one go.
// @match        https://github.com/*
// @icon         https://github.githubassets.com/favicons/favicon.svg
// @grant        GM_xmlhttpRequest
// @grant        GM_getValue
// @grant        GM_setValue
// @grant        GM_deleteValue
// @grant        GM_registerMenuCommand
// @connect      api.github.com
// @run-at       document-idle
// ==/UserScript==

(function () {
  'use strict';
  if (window.top !== window.self) return;

  /* ───────────────────────── config ───────────────────────── */
  const API = 'https://api.github.com';
  const TOKEN_KEY = 'ghu_token';
  const MAX_FILE_BYTES = 100 * 1024 * 1024; // GitHub blob limit
  const CONCURRENCY = 4;
  const MAX_ROWS = 250; // rows rendered in the file list
  const JUNK = [/(^|\/)\.DS_Store$/i, /(^|\/)Thumbs\.db$/i, /(^|\/)desktop\.ini$/i, /(^|\/)node_modules\//];
  const STEPS = ['Reading branch', 'Uploading files', 'Building tree', 'Creating commit', 'Updating branch'];
  const RESERVED = new Set(['settings', 'orgs', 'marketplace', 'explore', 'notifications', 'topics', 'sponsors', 'login', 'join', 'pulls', 'issues', 'codespaces', 'features', 'pricing', 'about', 'enterprise', 'apps', 'users', 'organizations', 'collections', 'trending', 'search', 'new', 'import', 'sessions', 'security', 'contact', 'customer-stories', 'readme', 'site', 'stars', 'watching', 'account', 'dashboard', 'copilot', 'solutions', 'resources', 'team', 'models', 'mcp', 'github-copilot']);

  /* ───────────────────────── tiny helpers ───────────────────────── */
  const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
  const encRef = (s) => s.split('/').map(encodeURIComponent).join('/');
  const fmt = (b) => (b < 1024 ? `${b} B` : b < 1048576 ? `${(b / 1024).toFixed(1)} KB` : b < 1073741824 ? `${(b / 1048576).toFixed(1)} MB` : `${(b / 1073741824).toFixed(2)} GB`);
  const cleanPath = (p) => (p || '').replace(/\\/g, '/').split('/').map((s) => s.trim()).filter(Boolean).join('/');
  const validPath = (p) => !p.split('/').some((s) => s === '.' || s === '..' || s.toLowerCase() === '.git');
  const plural = (n, w) => `${n} ${w}${n === 1 ? '' : 's'}`;
  const isJunk = (p) => JUNK.some((r) => r.test(p));

  const h = (tag, props, ...kids) => {
    const el = document.createElement(tag);
    for (const [k, v] of Object.entries(props || {})) {
      if (k === 'class') el.className = v;
      else if (k.startsWith('on')) el.addEventListener(k.slice(2).toLowerCase(), v);
      else if (v === true) el.setAttribute(k, '');
      else if (v !== false && v != null) el.setAttribute(k, v);
    }
    for (const kid of kids.flat(Infinity)) if (kid != null && kid !== false) el.append(kid.nodeType ? kid : document.createTextNode(kid));
    return el;
  };

  const ICON_PATHS = {
    upload: '<path d="M12 16V4M7 9l5-5 5 5M4 20h16"/>',
    folder: '<path d="M3 7a2 2 0 0 1 2-2h4l2 2h8a2 2 0 0 1 2 2v8a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2V7z"/>',
    folderPlus: '<path d="M3 7a2 2 0 0 1 2-2h4l2 2h8a2 2 0 0 1 2 2v8a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2V7z"/><path d="M12 10.5v5M9.5 13h5"/>',
    file: '<path d="M14 3H7a2 2 0 0 0-2 2v14a2 2 0 0 0 2 2h10a2 2 0 0 0 2-2V8l-5-5z"/><path d="M14 3v5h5"/>',
    x: '<path d="M6 6l12 12M18 6L6 18"/>',
    key: '<circle cx="8" cy="15" r="4"/><path d="M10.8 12.2L20 3M16 7l3 3"/>',
    check: '<path d="M5 12.5l4.5 4.5L19 7.5"/>',
    alert: '<circle cx="12" cy="12" r="9"/><path d="M12 8v5M12 16.5v.01"/>',
    branch: '<circle cx="6" cy="5" r="2"/><circle cx="6" cy="19" r="2"/><circle cx="18" cy="8" r="2"/><path d="M6 7v10M18 10c0 4-6 3-12 7"/>',
  };
  const icon = (name, size = 16) => {
    const s = h('span', { class: 'ico' });
    s.innerHTML = `<svg viewBox="0 0 24 24" width="${size}" height="${size}" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true">${ICON_PATHS[name]}</svg>`;
    return s;
  };

  /* ───────────────────────── state ───────────────────────── */
  const S = {
    open: false,
    view: 'main', // main | token | progress | done | error
    repo: null, // {owner, name, full}
    branch: '',
    defaultBranch: '',
    path: '',
    branches: [],
    locKey: '',
    locLoading: false,
    locError: '',
    tab: 'upload', // upload | folder
    folder: '',
    files: [], // [{path, file}]
    skipJunk: true,
    message: '',
    mode: 'direct', // direct | new
    newBranch: '',
    notice: null, // {type: info|warn|err, text}
    focus: null,
    token: { msg: '', ok: false, busy: false },
    progress: { i: 0, frac: 0, detail: '', note: '', total: 0 },
    result: null,
    error: '',
  };

  /* ───────────────────────── GitHub API ───────────────────────── */
  let onWait = null;

  function gm(method, url, body, token) {
    return new Promise((resolve, reject) => {
      GM_xmlhttpRequest({
        method,
        url,
        timeout: 180000,
        headers: Object.assign(
          { Accept: 'application/vnd.github+json', 'X-GitHub-Api-Version': '2022-11-28' },
          token ? { Authorization: `Bearer ${token}` } : {},
          body ? { 'Content-Type': 'application/json' } : {}
        ),
        data: body ? JSON.stringify(body) : undefined,
        onload: (r) => {
          let data = null;
          try { data = JSON.parse(r.responseText); } catch (_) { /* not json */ }
          resolve({ status: r.status, data, headers: r.responseHeaders || '' });
        },
        onerror: () => reject(new Error('Network error. Check your connection and try again.')),
        ontimeout: () => reject(new Error('GitHub took too long to respond.')),
      });
    });
  }

  async function api(method, path, body, opts = {}) {
    const token = GM_getValue(TOKEN_KEY, '');
    if (!token) { const e = new Error('No token saved.'); e.status = 401; throw e; }
    const retries = opts.retries == null ? 4 : opts.retries;
    for (let attempt = 0; ; attempt++) {
      const r = await gm(method, API + path, body, token);
      if (r.status >= 200 && r.status < 300) return r.data;
      const msg = (r.data && r.data.message) || `HTTP ${r.status}`;
      const limited = r.status === 429 || (r.status === 403 && /rate limit|abuse|secondary/i.test(msg));
      if (limited && attempt < retries) {
        const ra = /retry-after:\s*(\d+)/i.exec(r.headers);
        const secs = ra ? +ra[1] : Math.pow(2, attempt) * 4;
        if (onWait) onWait(secs);
        await sleep(secs * 1000);
        continue;
      }
      const err = new Error(msg);
      err.status = r.status;
      throw err;
    }
  }

  const friendly = (e) => {
    if (e.status === 401) return 'GitHub rejected your token (expired or revoked). Open the key icon and add a new one.';
    if (e.status === 403 || e.status === 404) return `${e.message}. Check that your token can write to this repo (classic token: "repo" scope; fine-grained: Contents → Read and write).`;
    if (e.status === 422 && /already exists/i.test(e.message)) return 'A branch with that name already exists. Pick a different name.';
    if (e.status === 422 && /fast forward/i.test(e.message)) return 'The branch changed while uploading. Try again.';
    return e.message || 'Something went wrong.';
  };

  /* ───────────────────────── page / repo detection ───────────────────────── */
  function parsePage() {
    const parts = location.pathname.split('/').filter(Boolean);
    if (parts.length < 2 || RESERVED.has(parts[0].toLowerCase())) return null;
    if (parts.length > 2 && !['tree', 'blob'].includes(parts[2])) return null;
    let rest = [];
    try { rest = parts.slice(3).map(decodeURIComponent); } catch (_) { rest = parts.slice(3); }
    return { owner: parts[0], name: parts[1], kind: parts[2] || null, rest };
  }

  async function loadLocation() {
    const page = parsePage();
    if (!page) { S.repo = null; return; }
    S.locLoading = true;
    S.locError = '';
    S.locKey = location.pathname;
    render();
    try {
      const base = `/repos/${page.owner}/${page.name}`;
      const info = await api('GET', base);
      let branch = info.default_branch;
      let parts = [];
      if (page.rest.length) {
        // A branch can contain slashes, and git forbids "a" and "a/b" coexisting, so the first match is exact.
        for (let i = 1; i <= page.rest.length; i++) {
          const cand = page.rest.slice(0, i).join('/');
          try {
            await api('GET', `${base}/git/ref/heads/${encRef(cand)}`, null, { retries: 0 });
            branch = cand;
            parts = page.rest.slice(i);
            if (page.kind === 'blob') parts.pop();
            break;
          } catch (e) {
            if (e.status === 401) throw e;
          }
        }
      }
      S.repo = { owner: page.owner, name: page.name, full: info.full_name };
      S.defaultBranch = info.default_branch;
      S.branch = branch;
      S.path = parts.join('/');
      if (info.permissions && info.permissions.push === false) {
        S.notice = { type: 'warn', text: `You don't have write access to ${info.full_name}. Fork it first, or use a token that has access.` };
      }
      S.locLoading = false;
      render();
      api('GET', `${base}/branches?per_page=100`).then((bs) => {
        S.branches = bs.map((b) => b.name);
        const dl = shadow.getElementById('ghu-branches');
        if (dl) dl.replaceChildren(...S.branches.map((b) => h('option', { value: b })));
      }).catch(() => {});
    } catch (e) {
      S.locLoading = false;
      S.locError = friendly(e);
      S.repo = null;
      if (e.status === 401) S.view = 'token';
      render();
    }
  }

  /* ───────────────────────── reading files ───────────────────────── */
  async function readEntry(entry, prefix) {
    const out = [];
    if (entry.isFile) {
      const file = await new Promise((res, rej) => entry.file(res, rej));
      out.push({ path: prefix + entry.name, file });
    } else if (entry.isDirectory) {
      const reader = entry.createReader();
      let batch;
      do {
        batch = await new Promise((res, rej) => reader.readEntries(res, rej));
        for (const child of batch) out.push(...(await readEntry(child, `${prefix}${entry.name}/`)));
      } while (batch.length);
    }
    return out;
  }

  async function filesFromDrop(dt) {
    // webkitGetAsEntry must be called synchronously during the drop event.
    const entries = Array.from(dt.items || []).filter((i) => i.kind === 'file').map((i) => (i.webkitGetAsEntry ? i.webkitGetAsEntry() : null)).filter(Boolean);
    if (!entries.length) return Array.from(dt.files || []).map((f) => ({ path: f.name, file: f }));
    const all = [];
    for (const e of entries) all.push(...(await readEntry(e, '')));
    return all;
  }

  const filesFromInput = (fileList) => Array.from(fileList).map((f) => ({ path: f.webkitRelativePath || f.name, file: f }));

  function addFiles(list) {
    const skipped = { junk: 0, big: 0, git: 0 };
    const map = new Map(S.files.map((f) => [f.path, f]));
    let added = 0;
    for (const { path, file } of list) {
      const p = cleanPath(path);
      if (!p) continue;
      if (!validPath(p)) { skipped.git++; continue; }
      if (S.skipJunk && isJunk(p)) { skipped.junk++; continue; }
      if (file.size > MAX_FILE_BYTES) { skipped.big++; continue; }
      if (!map.has(p)) added++;
      map.set(p, { path: p, file });
    }
    S.files = Array.from(map.values()).sort((a, b) => a.path.localeCompare(b.path));
    const notes = [];
    if (skipped.big) notes.push(`${plural(skipped.big, 'file')} over 100 MB`);
    if (skipped.git) notes.push(`${plural(skipped.git, 'item')} inside .git or with invalid names`);
    if (skipped.junk) notes.push(`${plural(skipped.junk, 'junk file')}`);
    if (notes.length) S.notice = { type: skipped.big || skipped.git ? 'warn' : 'info', text: `Skipped ${notes.join(', ')}.` };
    else S.notice = added ? null : S.notice;
  }

  async function handleDrop(dt) {
    const pending = filesFromDrop(dt); // starts synchronously
    S.notice = { type: 'info', text: 'Reading files…' };
    render();
    try {
      addFiles(await pending);
    } catch (_) {
      S.notice = { type: 'err', text: 'Could not read the dropped items.' };
    }
    render();
  }

  const toB64 = (file) =>
    new Promise((res, rej) => {
      const fr = new FileReader();
      fr.onload = () => res(String(fr.result).split(',')[1] || '');
      fr.onerror = () => rej(fr.error || new Error(`Could not read ${file.name}`));
      fr.readAsDataURL(file);
    });

  async function pool(items, n, worker) {
    let i = 0;
    let failed = false;
    const run = async () => {
      while (i < items.length && !failed) {
        const idx = i++;
        try { await worker(items[idx], idx); } catch (e) { failed = true; throw e; }
      }
    };
    await Promise.all(Array.from({ length: Math.min(n, items.length) }, run));
  }

  /* ───────────────────────── commit logic ───────────────────────── */
  function targetPrefix() {
    const parts = [];
    const base = cleanPath(S.path);
    if (base) parts.push(base);
    if (S.tab === 'folder') { const f = cleanPath(S.folder); if (f) parts.push(f); }
    return parts.join('/');
  }

  function buildItems() {
    const prefix = targetPrefix();
    const j = (p) => (prefix ? `${prefix}/${p}` : p);
    if (S.tab === 'folder' && !S.files.length) return [{ path: j('.gitkeep'), file: null }];
    return S.files.map((f) => ({ path: j(f.path), file: f.file }));
  }

  function defaultMessage() {
    const n = S.files.length;
    if (S.tab === 'folder') {
      const f = cleanPath(S.folder) || 'folder';
      return n ? `Add ${f} with ${plural(n, 'file')}` : `Create ${f} folder`;
    }
    if (n === 0) return 'Add files';
    if (n === 1) return `Add ${S.files[0].path.split('/').pop()}`;
    return `Add ${n} files`;
  }

  function validate() {
    if (!S.repo) return { ok: false, reason: S.locLoading ? 'Detecting repository…' : S.locError || 'Open a repository first' };
    if (!S.branch.trim()) return { ok: false, reason: 'Enter a branch name' };
    const base = cleanPath(S.path);
    if (base && !validPath(base)) return { ok: false, reason: 'Destination folder has a name GitHub does not allow' };
    if (S.tab === 'folder') {
      const f = cleanPath(S.folder);
      if (!f) return { ok: false, reason: 'Give the new folder a name' };
      if (!validPath(f)) return { ok: false, reason: 'That folder name is not allowed' };
    } else if (!S.files.length) {
      return { ok: false, reason: 'Add some files to get started' };
    }
    if (S.mode === 'new') {
      const b = S.newBranch.trim();
      if (!/^\w[\w.\-/]*$/.test(b) || /\.\.|\/\/|\/$|\.lock$/.test(b)) return { ok: false, reason: 'Enter a valid name for the new branch' };
    }
    return { ok: true };
  }

  function setStep(i, frac = 0, detail = '') {
    Object.assign(S.progress, { i, frac, detail, note: '' });
    updateProgress();
  }

  async function startCommit() {
    if (!validate().ok) return;
    const items = buildItems();
    const msg = S.message.trim() || defaultMessage();
    const { owner, name } = S.repo;
    const base = `/repos/${owner}/${name}`;
    const branch = S.branch.trim();
    const mode = S.mode;
    const newBranch = S.newBranch.trim();

    S.view = 'progress';
    S.progress = { i: 0, frac: 0, detail: '', note: '', total: items.length };
    render();
    onWait = (secs) => { S.progress.note = `GitHub asked us to slow down. Continuing in ${secs}s…`; updateProgress(); };

    try {
      setStep(0);
      let ref;
      try {
        ref = await api('GET', `${base}/git/ref/heads/${encRef(branch)}`);
      } catch (e) {
        if (e.status === 404) throw new Error(`The branch "${branch}" doesn't exist in ${owner}/${name}.`);
        if (e.status === 409) throw new Error('This repository is empty. Create a first commit on GitHub (for example a README), then try again.');
        throw e;
      }
      const headSha = ref.object.sha;
      const head = await api('GET', `${base}/git/commits/${headSha}`);

      setStep(1, 0, `0 / ${items.length}`);
      const entries = new Array(items.length);
      let done = 0;
      await pool(items, CONCURRENCY, async (it, idx) => {
        const content = it.file ? await toB64(it.file) : '';
        const blob = await api('POST', `${base}/git/blobs`, { content, encoding: 'base64' });
        entries[idx] = { path: it.path, mode: '100644', type: 'blob', sha: blob.sha };
        done++;
        setStep(1, done / items.length, `${done} / ${items.length}`);
      });

      setStep(2);
      const tree = await api('POST', `${base}/git/trees`, { base_tree: head.tree.sha, tree: entries });
      setStep(3);
      const commit = await api('POST', `${base}/git/commits`, { message: msg, tree: tree.sha, parents: [headSha] });
      setStep(4);
      let finalBranch = branch;
      if (mode === 'new') {
        await api('POST', `${base}/git/refs`, { ref: `refs/heads/${newBranch}`, sha: commit.sha });
        finalBranch = newBranch;
      } else {
        await api('PATCH', `${base}/git/refs/heads/${encRef(branch)}`, { sha: commit.sha });
      }
      setStep(5);

      S.result = { sha: commit.sha, branch: finalBranch, mode, count: items.length, owner, name, folder: S.tab === 'folder' && !S.files.length };
      S.files = [];
      S.folder = '';
      S.message = '';
      S.notice = null;
      S.view = 'done';
    } catch (e) {
      S.error = friendly(e);
      S.view = 'error';
    } finally {
      onWait = null;
    }
    render();
  }

  function tokenView() {
    const hasToken = !!GM_getValue(TOKEN_KEY, '');
    const inp = input({ type: 'password', 'data-f': 'token', placeholder: 'github_pat_… or ghp_…', onKeydown: (e) => { if (e.key === 'Enter') save(); } });
    const save = async () => {

      if (!t) return;
      S.token = { msg: 'Checking token…', ok: false, busy: true };
      render();
      const r = await gm('GET', `${API}/user`, null, t).catch((e) => ({ status: 0, data: { message: e.message } }));
      if (r.status === 200) {
        GM_setValue(TOKEN_KEY, t);
        S.token = { msg: '', ok: true, busy: false };
        S.view = 'main';
        S.notice = { type: 'info', text: `Connected as ${r.data.login}.` };
        render();
        loadLocation();
      } else {
        S.token = { msg: r.status === 401 ? 'GitHub says this token is not valid.' : (r.data && r.data.message) || 'Could not verify the token.', ok: false, busy: false };
        S.focus = 'token';
        render();
      }
    };
    const body = h('div', { class: 'body' },
      h('div', { class: 'explain' },
        h('div', {}, 'Uploads go through the GitHub API, so the script needs a personal access token. It is stored only in Tampermonkey on this browser and sent only to ', h('b', {}, 'api.github.com'), '.'),

  function openPanel() {

    if (S.open || !isFileDrag(e) || !parsePage()) return;
    if (e.target && e.target.closest && e.target.closest('textarea, input, [contenteditable="true"], file-attachment')) return;
    const p = parsePage();
    veilText.textContent = `Drop to add to ${p.owner}/${p.name}`;
    veil.classList.add('show');
  });
  veil.addEventListener('dragover', (e) => { e.preventDefault(); e.dataTransfer.dropEffect = 'copy'; });
  veil.addEventListener('dragleave', (e) => { if (!e.relatedTarget) hideVeil(); });
  veil.addEventListener('drop', (e) => {
    e.preventDefault();
    hideVeil();
    openPanel();
    if (S.view === 'main') handleDrop(e.dataTransfer);
  });
  window.addEventListener('dragend', hideVeil);
  window.addEventListener('keydown', (e) => { if (e.key === 'Escape') { hideVeil(); if (S.open) closePanel(); } });

  /* ───────────────────────── show / hide button on repo pages ───────────────────────── */
  function tick() { fab.classList.toggle('show', !!parsePage() && !S.open); }
  setInterval(tick, 700);
  ['turbo:load', 'turbo:render', 'pjax:end', 'popstate'].forEach((ev) => document.addEventListener(ev, tick));
  tick();

  /* ───────────────────────── Tampermonkey menu ───────────────────────── */
  if (typeof GM_registerMenuCommand === 'function') {
    GM_registerMenuCommand('Open repo uploader', openPanel);
    GM_registerMenuCommand('Set or change GitHub token', () => { if (!parsePage()) { alert('Open a repository page first.'); return; } S.open = true; S.view = 'token'; S.token = { msg: '', ok: false, busy: false }; S.focus = 'token'; render(); });
    GM_registerMenuCommand('Remove saved token', () => { GM_deleteValue(TOKEN_KEY); alert('Token removed.'); });
  }
})();
